local friendCache = {}
local function IsFriendCached(userId)
    if not userId then return false end
    if friendCache[userId] ~= nil then
        return friendCache[userId]
    end
    task.spawn(function()
        local f = false
        pcall(function()
            f = LocalPlayer:IsFriendsWith(userId)
        end)
        friendCache[userId] = f
    end)
    return false
end

local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name]
       or Settings.Whitelist.Players[player.UserId]
       or Settings.Whitelist.Players[tostring(player.UserId)]
       or Settings.Whitelist.Players[string.lower(player.Name)] then
        return true
    end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
        return true
    end
    if Settings.Whitelist.Friends then
        if IsFriendCached(player.UserId) then return true end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not humanoid or humanoid.Health <= 0 or not hrp then return false end
    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end
    return true
end

local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "HumanoidRootPart"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part and (partName == "Head" or partName == "UpperTorso") then
        part = character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
    return part or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
end

local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if usePred then
        local char = part.Parent
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local vel = (hrp and (hrp.AssemblyLinearVelocity or hrp.Velocity)) or part.AssemblyLinearVelocity or part.Velocity
        if vel then
            pos = pos + (vel * (predAmount or 0.135))
        end
    end
    return pos
end

local function GetClosestPlayerForAim()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local closest = nil
    local shortestDist = Settings.Aimbot.FOV or 220
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        closest = player
                    end
                end
            end
        end
    end
    return closest
end

        callback(active)

        Button = btn,
        Set = function(self, val)
            active = val
            btn.BackgroundColor3 = active and Color3.fromRGB(110, 25, 200) or Color3.fromRGB(20, 15, 32)
            btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
            btn.Text = text .. (active and ": ON" or ": OFF")
            callback(active)
        end
    }
end

local function CreateCycle(parent, label, options, defaultIndex, callback)
    local idx = defaultIndex or 1
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 11
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(90, 40, 160)
    stroke.Thickness = 1.2
    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
        callback(options[idx])
    end)
    return {
        Button = btn,
        Get = function() return options[idx] end,
        Set = function(self, val)
            for i, opt in ipairs(options) do
                if opt == val then
                    idx = i
                    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
                    callback(options[idx])
                    break
                end
            end
        end
    }
end

local updatingFromFloat = false

local currentFlyCFrame = nil

local function DoVClip(distance)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return end
    hrp.CFrame = hrp.CFrame + Vector3.new(0, distance, 0)
    hrp.AssemblyLinearVelocity = Vector3.zero
    if currentFlyCFrame then
        currentFlyCFrame = currentFlyCFrame + Vector3.new(0, distance, 0)
    end
end

local function UpdateFlyState(state, fromUI)
    Settings.Misc.Fly = state
    if FlyFloatBtn then
        if state then
            FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(200, 60, 30)
            FlyFloatBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            FlyFloatBtn.Text = "🚀 FLY: ON"
            flyFloatStroke.Color = Color3.fromRGB(255, 120, 60)
        else
            FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(24, 18, 22)
            FlyFloatBtn.TextColor3 = Color3.fromRGB(240, 180, 160)
            FlyFloatBtn.Text = "🚀 FLY: OFF"
            flyFloatStroke.Color = Color3.fromRGB(160, 60, 40)
        end
    end
    if not fromUI and flyToggleObj and flyToggleObj.Set then
        flyUpdating = true
        flyToggleObj:Set(state)
        flyUpdating = false
    end

    if state then
        if flyConnection then return end
        local startChar = LocalPlayer.Character
        local startHrp = startChar and startChar:FindFirstChild("HumanoidRootPart")
        if startHrp then
            currentFlyCFrame = startHrp.CFrame
        end

        flyConnection = RunService.Heartbeat:Connect(function(dt)
            if not Settings.Misc.Fly then
                if flyConnection then
                    flyConnection:Disconnect()
                    flyConnection = nil
                end
                return
            end

            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not hrp or not hum or hum.Health <= 0 then return end

            local cam = workspace.CurrentCamera or Camera
            local camCF = cam and cam.CFrame
            if not camCF then return end

            if not currentFlyCFrame then
                currentFlyCFrame = hrp.CFrame
            end

            for _, part in ipairs(char:GetChildren()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end

            hrp.AssemblyLinearVelocity = Vector3.zero
            local moveVector = Vector3.zero

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                moveVector = moveVector + camCF.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                moveVector = moveVector - camCF.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                moveVector = moveVector - camCF.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                moveVector = moveVector + camCF.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                moveVector = moveVector + Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveVector = moveVector - Vector3.new(0, 1, 0)
            end

            -- на телефоне работает от стандартного джойстика Roblox
            if moveVector.Magnitude == 0 and hum.MoveDirection.Magnitude > 0 then
                local moveDir = hum.MoveDirection
                moveVector = moveVector + Vector3.new(moveDir.X, camCF.LookVector.Y * moveDir.Magnitude, moveDir.Z)
            end

            if moveVector.Magnitude > 0 then
                moveVector = moveVector.Unit
                local speed = Settings.Misc.FlySpeed or 50
                local deltaTime = dt or 0.016
                currentFlyCFrame = currentFlyCFrame + (moveVector * (speed * deltaTime))
            end

            hrp.CFrame = CFrame.new(currentFlyCFrame.Position) * camCF.Rotation
            hrp.AssemblyLinearVelocity = Vector3.zero
        end)
        table.insert(_G.CociteConnections, flyConnection)
    else
        if flyConnection then
            flyConnection:Disconnect()
            flyConnection = nil
        end
        currentFlyCFrame = nil
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end
end

-- как на видео: авто-лок ближайшего в круге
local function GetClosestSilentTarget()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    local shortestDist = Settings.SilentAim.FOV
    local bestTarget = nil
    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.SilentAim.AimPart)
            if part then
                local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        if not Settings.SilentAim.WallCheck or IsWallCheckVisible(part, player.Character) then
                            shortestDist = dist
                            bestTarget = player
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

-- [[ WALL CHECK ]]
local isInternalRay = false

IsWallCheckVisible = function(part, targetChar)
    if not Settings.SilentAim.WallCheck then return true end
    local localChar = LocalPlayer.Character
    if not localChar then return false end

    local origin = Camera.CFrame.Position
    local dest = part.Position
    local dir = dest - origin

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {localChar, Camera}
    params.IgnoreWater = true

    isInternalRay = true
    local hit = workspace:Raycast(origin, dir, params)
    isInternalRay = false
    if not hit or hit.Instance:IsDescendantOf(targetChar) then
        return true
    end
    return false
end

-- [[ NO SPREAD ENGINE ]]
pcall(function()
    if hookfunction and newcclosure and checkcaller then
        local oldRandom
        oldRandom = hookfunction(math.random, newcclosure(function(...)
            local args = {...}
            if checkcaller() then
                return oldRandom(...)
            end
            if Settings.NoSpread and Settings.NoSpread.Enabled then
                if (#args == 0) or
                   (args[1] == -0.05 and args[2] == 0.05) or
                   (args[1] == -0.1 and args[2] == 0.1) or
                   (args[1] == -0.1) or
                   (args[1] == -0.05) then
                    local spread = Settings.NoSpread.Amount or 0
                    return oldRandom(...) * (spread / 100)
                end
            end
            return oldRandom(...)
        end))
    end
end)

local function CalcRayDirection(origin, targetPos, spread)
    local dir = (targetPos - origin).Unit
    if spread and spread > 0 then
        local spreadDist = spread * ((targetPos - origin).Magnitude / 100)
        dir = ((targetPos + Vector3.new(
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000
        )) - origin).Unit
    end
    return dir * 1000
end

local function GetStompTool()
    local char = LocalPlayer.Character
    if not char then return nil end
    local stomp = char:FindFirstChild("Stomp")
    if stomp then return stomp end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    stomp = bp and bp:FindFirstChild("Stomp")
    if stomp then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum:EquipTool(stomp)
            return char:FindFirstChild("Stomp")
        end
    end
    return nil
end

    if not killTarget then return end

    local tChar = killTarget.Character
    local tHum = tChar and tChar:FindFirstChildOfClass("Humanoid")
    local tEffects = tChar and tChar:FindFirstChild("BodyEffects")
    local ko = tEffects and (tEffects:FindFirstChild("K.O") or tEffects:FindFirstChild("KO"))
    local deadFlag = tEffects and tEffects:FindFirstChild("Dead")
    local koNow = ko and ko.Value == true
    local isDead = (not tChar) or (deadFlag and deadFlag.Value == true) or (tHum and tHum.Health <= 0 and not koNow)

    if isDead then
        killTarget = nil
        killStomping = false
        if RefreshKillList then RefreshKillList() end
        return
    end

    if koNow then
        local lChar = LocalPlayer.Character
        local lHrp = lChar and lChar:FindFirstChild("HumanoidRootPart")
        local tRoot = tChar:FindFirstChild("HumanoidRootPart") or tChar:FindFirstChild("UpperTorso") or tChar:FindFirstChild("Torso")
        if lHrp and tRoot and (lHrp.Position - tRoot.Position).Magnitude <= 12 then
            killStomping = true
            local now = tick()
            if now - lastStomp >= 0.15 then
                lastStomp = now
                local stomp = GetStompTool()
                if stomp then
                    pcall(function() stomp:Activate() end)
                end
            end
        else
            killStomping = false
        end
    elseif killStomping then
        killStomping = false
    end
end))

        local aimTarget = killTarget or lockedSilentTarget
        if not aimTarget and Settings.SilentAim.AutoLock then
            aimTarget = GetClosestSilentTarget()
        end

        if aimTarget and IsValidTarget(aimTarget) then
            local part = GetCharacterAimPart(aimTarget.Character, Settings.SilentAim.AimPart)
            if part then
                -- sticky: залоченная/выбранная цель редиректит всегда, без проверки FOV и экрана
                if not Settings.SilentAim.WallCheck or IsWallCheckVisible(part, aimTarget.Character) then
                    bestTarget = aimTarget
                    bestPart = part
                    bestPos = GetPredictedPosition(part, Settings.SilentAim.UsePrediction, Settings.SilentAim.Prediction)
                end
            end
        end

        if lockedSilentTarget and (not IsValidTarget(lockedSilentTarget)) then
            lockedSilentTarget = nil
            if not killTarget then
                SilentFloatBtn.BackgroundColor3 = Color3.fromRGB(0, 130, 180)
                SilentFloatBtn.Text = "🔮 BULLET TP: ON"
            end
        end
    end

    SetLockTag(bestTarget)

    currentSilentTarget = bestTarget
    currentSilentTargetPart = bestPart
    currentSilentAimPos = bestPos

    if AimbotFOVCircle then
        AimbotFOVCircle.Visible = Settings.Aimbot.Enabled and Settings.Aimbot.ShowFOV
        AimbotFOVCircle.Radius = Settings.Aimbot.FOV
        AimbotFOVCircle.Position = center
    end
    if SilentFOVCircle then
        SilentFOVCircle.Visible = Settings.SilentAim.Enabled and Settings.SilentAim.ShowFOV
        SilentFOVCircle.Radius = Settings.SilentAim.FOV
        SilentFOVCircle.Position = center
    end

    if SilentTracerLine then
        if Settings.SilentAim.Enabled and Settings.SilentAim.Tracers and currentSilentTargetPart then
            local screenPos, onScreen = cam:WorldToViewportPoint(currentSilentTargetPart.Position)
            if onScreen and screenPos.Z > 0 then
                SilentTracerLine.From = center
                SilentTracerLine.To = Vector2.new(screenPos.X, screenPos.Y)
                SilentTracerLine.Visible = true
            else
                SilentTracerLine.Visible = false
            end
        else
            SilentTracerLine.Visible = false
        end
    end
end)

-- Raycast + UpdateMousePos redirect
pcall(function()
    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            local self = args[1]

            if not checkcaller() and not isInternalRay and Settings.SilentAim.Enabled and currentSilentAimPos then
                local willHit = true
                if Settings.SilentAim.HitChance < 100 then
                    willHit = (math.random(1, 100) <= Settings.SilentAim.HitChance)
                end

                if willHit then
                    if self == workspace and method == "Raycast" then
                        if typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                            local origin = args[2]
                            local targetPos = currentSilentAimPos

                            local toolName = equippedToolName or ""
                            local isShotgun = string.find(string.lower(toolName), "double")
                                           or string.find(string.lower(toolName), "tactical")
                                           or string.find(string.lower(toolName), "shotgun")
                            local spreadVal = 0
                            if not (Settings.NoSpread and Settings.NoSpread.Enabled) then
                                spreadVal = isShotgun and 6 or 0
                            else
                                spreadVal = ((Settings.NoSpread.Amount or 0) / 100) * (isShotgun and 6 or 0)
                            end
                            args[3] = CalcRayDirection(origin, targetPos, spreadVal)
                            return oldNamecall(table.unpack(args))
                        end
                    elseif method == "FireServer" and tostring(self) == "MainEvent" then
                        local eventArg = tostring(args[2] or args[1] or "")
                        if eventArg == "UpdateMousePos" or eventArg == "UpdateMousePosI" or string.find(string.lower(eventArg), "mousepos") then
                            if args[2] == eventArg then
                                args[3] = currentSilentAimPos
                            else
                                args[2] = currentSilentAimPos
                            end
                            return oldNamecall(table.unpack(args))
                        end
                    end
                end
            end

            return oldNamecall(...)
        end))
    end
end)

-- [[ ESP ENGINE ]]
local espData = {}

local function RemoveESP(player)
    local d = espData[player]
    if d then
        if d.Box then pcall(function() d.Box:Remove() end) end
        if d.BB then pcall(function() d.BB:Destroy() end) end
        if d.HL then pcall(function() d.HL:Destroy() end) end
        espData[player] = nil
    end
end

local function EnsureESP(player)
    if player == LocalPlayer or espData[player] then return end
    local d = {Box = nil, BB = nil, HL = nil}
    pcall(function()
        if Drawing and Drawing.new then
            local box = Drawing.new("Square")
            box.Thickness = 1.5
            box.Filled = false
            box.Visible = false
            box.Color = Color3.fromRGB(180, 60, 255)
            d.Box = box
        end
    end)
    espData[player] = d
end

for _, p in ipairs(Players:GetPlayers()) do EnsureESP(p) end
table.insert(_G.CociteConnections, Players.PlayerAdded:Connect(EnsureESP))
table.insert(_G.CociteConnections, Players.PlayerRemoving:Connect(RemoveESP))

table.insert(_G.CociteConnections, RunService.Heartbeat:Connect(function()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return end
    for player, d in pairs(espData) do
        local show = Settings.ESP.Enabled and IsValidTarget(player)
        if show then
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local head = char and char:FindFirstChild("Head")
            if hrp and head then
                -- chams
                if Settings.ESP.Chams then
                    if not d.HL or d.HL.Parent ~= char then
                        if d.HL then pcall(function() d.HL:Destroy() end) end
                        local hl = Instance.new("Highlight")
                        hl.FillColor = Color3.fromRGB(140, 40, 255)
                        hl.OutlineColor = Color3.fromRGB(220, 150, 255)
                        hl.FillTransparency = 0.55
                        hl.OutlineTransparency = 0
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Parent = char
                        d.HL = hl
                    end
                elseif d.HL then
                    pcall(function() d.HL:Destroy() end)
                    d.HL = nil
                end
                -- names
                if Settings.ESP.Names then
                    if not d.BB or d.BB.Parent ~= head then
                        if d.BB then pcall(function() d.BB:Destroy() end) end
                        local bb = Instance.new("BillboardGui")
                        bb.Size = UDim2.new(0, 100, 0, 18)
                        bb.StudsOffset = Vector3.new(0, 2, 0)
                        bb.AlwaysOnTop = true
                        local tl = Instance.new("TextLabel")
                        tl.Size = UDim2.new(1, 0, 1, 0)
                        tl.BackgroundTransparency = 1
                        tl.Font = Enum.Font.GothamBold
                        tl.Text = player.Name
                        tl.TextColor3 = Color3.fromRGB(255, 255, 255)
                        tl.TextSize = 12
                        tl.TextStrokeTransparency = 0.4
                        tl.Parent = bb
                        bb.Parent = head
                        d.BB = bb
                    end
                elseif d.BB then
                    pcall(function() d.BB:Destroy() end)
                    d.BB = nil
                end
                -- box
                if d.Box then
                    if Settings.ESP.Boxes then
                        local top = cam:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
                        local bottom = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                        if top.Z > 0 and bottom.Z > 0 then
                            local h = math.abs(top.Y - bottom.Y)
                            local w = h * 0.65
                            d.Box.Size = Vector2.new(w, h)
                            d.Box.Position = Vector2.new(top.X - w / 2, top.Y)
                            d.Box.Visible = true
                        else
                            d.Box.Visible = false
                        end
                    else
                        d.Box.Visible = false
                    end
                end
            else
                if d.Box then d.Box.Visible = false end
                if d.HL then pcall(function() d.HL:Destroy() end) d.HL = nil end
                if d.BB then pcall(function() d.BB:Destroy() end) d.BB = nil end
            end
        else
            if d.Box then d.Box.Visible = false end
            if d.HL then pcall(function() d.HL:Destroy() end) d.HL = nil end
            if d.BB then pcall(function() d.BB:Destroy() end) d.BB = nil end
        end
    end
end))
