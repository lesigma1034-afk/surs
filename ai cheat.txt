local function GetClosestTarget()
    local closest, shortest = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local head = p.Character:FindFirstChild("Head")
            if head then
                local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude
                    if dist < shortest then
                        shortest = dist
                        closest = p
                    end
                end
            end
        end
    end
    return closest
end

local AimActive = false
AimFloat.MouseButton1Click:Connect(function()
    AimActive = not AimActive
    if AimActive then
        AimFloat.BackgroundColor3 = Color3.fromRGB(90, 60, 200)
        AimFloat.TextColor3 = Color3.fromRGB(255, 255, 255)
        AimFloat.Text = "AIM: ON"
        Settings.Aimbot.Target = GetClosestTarget()
    else
        AimFloat.BackgroundColor3 = Color3.fromRGB(22, 22, 32)
        AimFloat.TextColor3 = Color3.fromRGB(180, 180, 200)
        AimFloat.Text = "AIM: OFF"
        Settings.Aimbot.Target = nil
    end
end)

-- 4. ЛОГИКА ЧИТОВ (RENDERSTEPPED)
-------------------------------------------------------------------
RunService.RenderStepped:Connect(function()
    -- Aimbot Lock
    if Settings.Aimbot.Enabled and AimActive then
        local target = Settings.Aimbot.Target
        if target and target.Character and target.Character:FindFirstChild("Head") and target.Character.Humanoid.Health > 0 then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Character.Head.Position)
        else
            Settings.Aimbot.Target = GetClosestTarget()
        end
    end
    
    -- Bullet TP (Перенос патрона/хендла к врагу)
    if Settings.BulletTP.Enabled then
        pcall(function()
            local target = GetClosestTarget()
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                local hrp = target.Character.HumanoidRootPart
                for _, v in pairs(workspace:GetChildren()) do
                    if v.Name == "Handle" and v:IsA("BasePart") then
                        v.CFrame = hrp.CFrame
                    end
                end
            end
        end)
    end
    
    -- Noclip
    if Settings.Noclip.Enabled and LocalPlayer.Character then
        for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end
end)

local function CreateNewButton(text, posY, callback)
    local btn = Instance.new("TextButton", MainHub)
    btn.BackgroundColor3 = Color3.fromRGB(22, 22, 32)
    btn.Position = UDim2.new(0, 20, 0, posY)
    btn.Size = UDim2.new(0, 310, 0, 46)
    btn.Font = Enum.Font.GothamMedium
    btn.Text = text .. " [OFF]"
    btn.TextColor3 = Color3.fromRGB(170, 170, 190)
    btn.TextSize = 13
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    
    local state = false
    btn.MouseButton1Click:Connect(function()
        state = not state
        if state then
            btn.BackgroundColor3 = Color3.fromRGB(80, 50, 180)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = text .. " [ON]"
        else
            btn.BackgroundColor3 = Color3.fromRGB(22, 22, 32)
            btn.TextColor3 = Color3.fromRGB(170, 170, 190)
            btn.Text = text .. " [OFF]"
        end
        callback(state)
    end)
    return btn
end

-- 6. ESP (BOXES + NAMES)
-------------------------------------------------------------------
local function ToggleESP(state)
    Settings.ESP.Enabled = state
    if not state then
        for _, box in pairs(Settings.ESP.Boxes) do box:Remove() end
        for _, name in pairs(Settings.ESP.Names) do name:Remove() end
        Settings.ESP.Boxes = {}
        Settings.ESP.Names = {}
    end
end

RunService.RenderStepped:Connect(function()
    if not Settings.ESP.Enabled then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local hrp = p.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            
            if not Settings.ESP.Boxes[p] then
                local box = Drawing.new("Square")
                box.Visible = false
                box.Color = Color3.fromRGB(150, 100, 255)
                box.Thickness = 1.5
                box.Filled = false
                Settings.ESP.Boxes[p] = box
            end
            
            if not Settings.ESP.Names[p] then
                local txt = Drawing.new("Text")
                txt.Visible = false
                txt.Color = Color3.fromRGB(255, 255, 255)
                txt.Size = 12
                txt.Center = true
                txt.Outline = true
                Settings.ESP.Names[p] = txt
            end
            
            local box = Settings.ESP.Boxes[p]
            local txt = Settings.ESP.Names[p]
            
            if onScreen then
                local size = Vector3.new(2, 3, 0) * (300 / pos.Z)
                box.Size = Vector2.new(size.X, size.Y)
                box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                box.Visible = true
                
                txt.Text = p.Name
                txt.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 16)
                txt.Visible = true
            else
                box.Visible = false
                txt.Visible = false
            end
        else
            if Settings.ESP.Boxes[p] then Settings.ESP.Boxes[p].Visible = false end
            if Settings.ESP.Names[p] then Settings.ESP.Names[p].Visible = false end
        end
    end
end)
