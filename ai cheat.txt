local function GetClosestTarget()
    local closestP = nil
    local shortestDist = math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            local part = player.Character:FindFirstChild("Head")
            if part then
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

        Settings.Aimbot.LockedTarget = GetClosestTarget()
    else

        AimFloatBtn.TextColor3 = Color3.fromRGB(210, 160, 255)

RunService.RenderStepped:Connect(function()
    -- Aimbot лок
    if Settings.Aimbot.Enabled and AimActive then
        local target = Settings.Aimbot.LockedTarget
        if target and target.Character and target.Character:FindFirstChild("Head") and target.Character.Humanoid.Health > 0 then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Character.Head.Position)
        else
            Settings.Aimbot.LockedTarget = GetClosestTarget()
        end
    end
    
    -- Bullet TP (Телепортация пуль/хитов к ближайшему врагу при стрельбе)
    if Settings.BulletTP.Enabled then
        pcall(function()
            local target = GetClosestTarget()
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                if tool and tool:FindFirstChild("Handle") then
                    -- Быстрое сближение хитбоксов патронов/оружия для регистрации урона на дальней дистанции
                    local hrp = target.Character.HumanoidRootPart
                    for _, v in pairs(workspace:GetDescendants()) do
                        if v.Name == "Handle" and v.Parent == workspace then
                            v.CFrame = hrp.CFrame
                        end
                    end
                end
            end
        end)
    end
    
    -- Noclip
    if Settings.Noclip.Enabled and LocalPlayer.Character then
        for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end
end)

local function ToggleESP(state)
    Settings.ESP.Enabled = state
    if not state then
        for _, box in pairs(Settings.ESP.Boxes) do box:Remove() end
        for _, name in pairs(Settings.ESP.Names) do name:Remove() end
        Settings.ESP.Boxes = {}
        Settings.ESP.Names = {}
    end
end

RunService.RenderStepped:Connect(function()
    if not Settings.ESP.Enabled then return end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            local hrp = player.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            
            if not Settings.ESP.Boxes[player] then
                local box = Drawing.new("Square")
                box.Visible = false
                box.Color = Color3.fromRGB(180, 50, 255)
                box.Thickness = 1.5
                box.Filled = false
                Settings.ESP.Boxes[player] = box
            end
            
            if not Settings.ESP.Names[player] then
                local nameText = Drawing.new("Text")
                nameText.Visible = false
                nameText.Color = Color3.fromRGB(255, 255, 255)
                nameText.Size = 13
                nameText.Center = true
                nameText.Outline = true
                Settings.ESP.Names[player] = nameText
            end
            
            local box = Settings.ESP.Boxes[player]
            local nameText = Settings.ESP.Names[player]
            
            if onScreen then
                local size = Vector3.new(2, 3, 0) * (300 / pos.Z)
                box.Size = Vector2.new(size.X, size.Y)
                box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                box.Visible = true
                
                nameText.Text = player.Name
                nameText.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 18)
                nameText.Visible = true
            else
                box.Visible = false
                nameText.Visible = false
            end
        else
            if Settings.ESP.Boxes[player] then Settings.ESP.Boxes[player].Visible = false end
            if Settings.ESP.Names[player] then Settings.ESP.Names[player].Visible = false end
        end
    end
end)
