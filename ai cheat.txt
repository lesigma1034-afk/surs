local function CreateToggle(name, yPos, callback)
    local btn = Instance.new("TextButton", MainFrame)
    btn.Size = UDim2.new(0, 280, 0, 45)
    btn.Position = UDim2.new(0, 15, 0, yPos)
    btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = name .. " [ OFF ]"
    btn.TextColor3 = Color3.fromRGB(190, 180, 210)
    btn.TextSize = 12
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 180)
    stroke.Thickness = 1.2

    local state = false
    btn.MouseButton1Click:Connect(function()
        state = not state
        if state then
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(130, 30, 230)}):Play()
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = name .. " [ VIP ON ]"
        else
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(20, 15, 32)}):Play()
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = name .. " [ OFF ]"
        end
        callback(state)
    end)
    return btn
end

local function GetTarget()
    local target, dist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            local head = p.Character:FindFirstChild("Head")
            if head then
                local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local mag = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude
                    if mag < dist then dist, target = mag, p end
                end
            end
        end
    end
    return target
end

CreateToggle("Aimbot", 55, function(state)
    Settings.Aimbot.Enabled = state
end)

CreateToggle("ESP (Boxes + Names)", 110, function(state)
    Settings.ESP.Enabled = state
    if not state then
        for _, v in pairs(Settings.ESP.Cache) do
            if v.Box then v.Box:Remove() end
            if v.Name then v.Name:Remove() end
        end
        Settings.ESP.Cache = {}
    end
end)

CreateToggle("Noclip", 165, function(state)
    Settings.Noclip.Enabled = state
end)

RunService.RenderStepped:Connect(function()
    if Settings.Aimbot.Enabled then
        local t = Settings.Aimbot.Target
        if not t or not t.Character or not t.Character:FindFirstChild("Head") or t.Character.Humanoid.Health <= 0 then
            Settings.Aimbot.Target = GetTarget()
            t = Settings.Aimbot.Target
        end
        if t and t.Character and t.Character:FindFirstChild("Head") then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, t.Character.Head.Position)
        end
    end

    if Settings.Noclip.Enabled and LocalPlayer.Character then
        for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end

    if Settings.ESP.Enabled then
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local char = p.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local hum = char and char:FindFirstChild("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if not Settings.ESP.Cache[p] then
                        local box = Drawing.new("Square")
                        box.Color = Color3.fromRGB(200, 60, 255)
                        box.Thickness = 1.5
                        box.Filled = false
                        local name = Drawing.new("Text")
                        name.Color = Color3.fromRGB(255, 255, 255)
                        name.Size = 12
                        name.Center = true
                        name.Outline = true
                        Settings.ESP.Cache[p] = {Box = box, Name = name}
                    end
                    local v = Settings.ESP.Cache[p]
                    if onScreen then
                        local scale = 300 / pos.Z
                        local size = Vector2.new(2, 3) * scale
                        v.Box.Size = size
                        v.Box.Position = Vector2.new(pos.X - size.X/2, pos.Y - size.Y/2)
                        v.Box.Visible = true
                        v.Name.Text = p.Name
                        v.Name.Position = Vector2.new(pos.X, pos.Y - size.Y/2 - 16)
                        v.Name.Visible = true
                    else
                        v.Box.Visible = false
                        v.Name.Visible = false
                    end
                else
                    if Settings.ESP.Cache[p] then
                        Settings.ESP.Cache[p].Box.Visible = false
                        Settings.ESP.Cache[p].Name.Visible = false
                    end
                end
            end
        end
    end
end)
