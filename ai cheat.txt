-- ============================================================
-- TARGET VALIDATION (with Whitelist protection)
-- ============================================================
local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end

    -- Whitelist check (Name + DisplayName, case-insensitive)
    local wl = Settings.Aimbot.Whitelist
    if wl then
        local lname = player.Name:lower()
        local dname = (player.DisplayName or ""):lower()
        if wl[lname] or wl[dname] then
            return false
        end
    end

    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not hrp then return false end

    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end

    return true
end

-- CREATE TOGGLE (helper)
-- ============================================================
local function CreateToggle(parent, text, defaultState, callback)
    local ToggleFrame = Instance.new("Frame", parent)
    ToggleFrame.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
    ToggleFrame.Size = UDim2.new(1, -20, 0, 34)
    ToggleFrame.BackgroundTransparency = 0.1
    Instance.new("UICorner", ToggleFrame).CornerRadius = UDim.new(0, 8)

    local Label = Instance.new("TextLabel", ToggleFrame)
    Label.BackgroundTransparency = 1
    Label.Position = UDim2.new(0, 12, 0, 0)
    Label.Size = UDim2.new(1, -70, 1, 0)
    Label.Font = Enum.Font.GothamSemibold
    Label.Text = text
    Label.TextColor3 = Color3.fromRGB(220, 210, 240)
    Label.TextSize = 12
    Label.TextXAlignment = Enum.TextXAlignment.Left

    local Btn = Instance.new("TextButton", ToggleFrame)
    Btn.BackgroundColor3 = defaultState and Color3.fromRGB(130, 35, 235) or Color3.fromRGB(40, 30, 60)
    Btn.Position = UDim2.new(1, -54, 0.5, -10)
    Btn.Size = UDim2.new(0, 42, 0, 20)
    Btn.Text = ""
    Btn.AutoButtonColor = false
    Instance.new("UICorner", Btn).CornerRadius = UDim.new(1, 0)

    local Dot = Instance.new("Frame", Btn)
    Dot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Dot.Size = UDim2.new(0, 16, 0, 16)
    Dot.Position = defaultState and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
    Instance.new("UICorner", Dot).CornerRadius = UDim.new(1, 0)

    local state = defaultState
    Btn.MouseButton1Click:Connect(function()
        state = not state
        TweenService:Create(Btn, TweenInfo.new(0.18), {
            BackgroundColor3 = state and Color3.fromRGB(130, 35, 235) or Color3.fromRGB(40, 30, 60)
        }):Play()
        TweenService:Create(Dot, TweenInfo.new(0.18), {
            Position = state and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
        }):Play()
        if callback then callback(state) end
    end)

    return ToggleFrame, Btn, Label
end

-- ============================================================
-- KILL AURA LOGIC (RenderStepped)
-- ============================================================
local lastActivate = 0

RunService.RenderStepped:Connect(function(dt)
    if not Settings.Aimbot.KillAura then return end

    local character = LocalPlayer.Character
    if not character then return end

    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local maxFov = Settings.Aimbot.AuraFOV or 150

    local closest, closestDist = nil, maxFov

    for _, plr in ipairs(Players:GetPlayers()) do
        if IsValidTarget(plr) then
            local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    if dist < closestDist then
                        closestDist = dist
                        closest = plr
                    end
                end
            end
        end
    end

    if closest then
        local hrp = closest.Character and closest.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            -- Aim camera at target
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, hrp.Position)

            -- Fire if a tool is equipped
            local tool = character:FindFirstChildOfClass("Tool")
            if tool then
                local now = tick()
                if now - lastActivate >= 0.05 then
                    lastActivate = now
                    pcall(function()
                        tool:Activate()
                    end)
                end
            end
        end
    end
end)
