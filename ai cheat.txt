local friendCache = {}
local function IsFriendCached(userId)
    if not userId then return false end
    if friendCache[userId] ~= nil then
        return friendCache[userId]
    end
    task.spawn(function()
        local f = false
        pcall(function()
            f = LocalPlayer:IsFriendsWith(userId)
        end)
        friendCache[userId] = f
    end)
    return false
end

local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name]
       or Settings.Whitelist.Players[player.UserId]
       or Settings.Whitelist.Players[tostring(player.UserId)]
       or Settings.Whitelist.Players[string.lower(player.Name)] then
        return true
    end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
        return true
    end
    if Settings.Whitelist.Friends then
        if IsFriendCached(player.UserId) then return true end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not hrp then return false end

    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end

    return true
end

local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "HumanoidRootPart"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part and (partName == "Head" or partName == "UpperTorso") then
        part = character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
    return part or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
end

local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if usePred then
        local char = part.Parent
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local vel = (hrp and (hrp.AssemblyLinearVelocity or hrp.Velocity)) or part.AssemblyLinearVelocity or part.Velocity
        if vel then
            pos = pos + (vel * (predAmount or 0.135))
        end
    end
    return pos
end

local function GetClosestPlayerForAim()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local closest = nil
    local shortestDist = Settings.Aimbot.FOV or 220
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        closest = player
                    end
                end
            end
        end
    end
    return closest
end

    btn.Text = text .. (defaultState and ": ON" or ": OFF")
    btn.TextColor3 = defaultState and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 170)
    stroke.Thickness = 1.3

    local active = defaultState
    btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(120, 30, 210)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = text .. ": ON"
        else
            btn.BackgroundColor3 = Color3.fromRGB(18, 14, 30)
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = text .. ": OFF"
        end
        callback(active)
    end)

    return {
        Button = btn,
        Set = function(self, val)
            active = val
            btn.BackgroundColor3 = active and Color3.fromRGB(120, 30, 210) or Color3.fromRGB(18, 14, 30)
            btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
            btn.Text = text .. (active and ": ON" or ": OFF")
            callback(active)
        end
    }
end

local function CreateCycle(parent, label, options, defaultIndex, callback)
    local idx = defaultIndex or 1
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = Color3.fromRGB(18, 14, 30)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 170)
    stroke.Thickness = 1.3

    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
        callback(options[idx])
    end)

    return {
        Button = btn,
        Get = function() return options[idx] end,
        Set = function(self, val)
            for i, opt in ipairs(options) do
                if opt == val then
                    idx = i
                    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
                    callback(options[idx])
                    break
                end
            end
        end
    }
end

local function DoVClip(distance)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return end

    hrp.CFrame = hrp.CFrame + Vector3.new(0, distance, 0)
    hrp.AssemblyLinearVelocity = Vector3.zero
    if currentFlyCFrame then
        currentFlyCFrame = currentFlyCFrame + Vector3.new(0, distance, 0)
    end
end

local function UpdateFlyState(state, fromUI)
    Settings.Misc.Fly = state
    if FlyFloatBtn then
        if state then
            FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(210, 65, 35)
            FlyFloatBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            FlyFloatBtn.Text = "🚀 FLY: ON"
            flyFloatStroke.Color = Color3.fromRGB(255, 130, 70)
        else
            FlyFloatBtn.BackgroundColor3 = Color3.fromRGB(22, 16, 20)
            FlyFloatBtn.TextColor3 = Color3.fromRGB(240, 180, 160)
            FlyFloatBtn.Text = "🚀 FLY: OFF"
            flyFloatStroke.Color = Color3.fromRGB(170, 70, 45)
        end
    end
    if not fromUI and flyToggleObj and flyToggleObj.Set then
        flyToggleObj:Set(state)
    end

    if state then
        if flyConnection then return end
        local startChar = LocalPlayer.Character
        local startHrp = startChar and startChar:FindFirstChild("HumanoidRootPart")
        if startHrp then
            currentFlyCFrame = startHrp.CFrame
        end

        flyConnection = RunService.Heartbeat:Connect(function(dt)
            if not Settings.Misc.Fly then
                if flyConnection then
                    flyConnection:Disconnect()
                    flyConnection = nil
                end
                return
            end

            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not hrp or not hum or hum.Health <= 0 then return end

            local cam = workspace.CurrentCamera or Camera
            local camCF = cam and cam.CFrame
            if not camCF then return end

            if not currentFlyCFrame then
                currentFlyCFrame = hrp.CFrame
            end

            for _, part in ipairs(char:GetChildren()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end

            hrp.AssemblyLinearVelocity = Vector3.zero

            local moveVector = Vector3.zero

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                moveVector = moveVector + camCF.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                moveVector = moveVector - camCF.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                moveVector = moveVector - camCF.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                moveVector = moveVector + camCF.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                moveVector = moveVector + Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveVector = moveVector - Vector3.new(0, 1, 0)
            end

            if moveVector.Magnitude == 0 and hum.MoveDirection.Magnitude > 0 then
                local moveDir = hum.MoveDirection
                moveVector = moveVector + Vector3.new(moveDir.X, camCF.LookVector.Y * moveDir.Magnitude, moveDir.Z)
            end

            if moveVector.Magnitude > 0 then
                moveVector = moveVector.Unit
                local speed = Settings.Misc.FlySpeed or 50
                local deltaTime = dt or 0.016
                currentFlyCFrame = currentFlyCFrame + (moveVector * (speed * deltaTime))
            end

            hrp.CFrame = CFrame.new(currentFlyCFrame.Position) * camCF.Rotation
            hrp.AssemblyLinearVelocity = Vector3.zero
        end)
        table.insert(_G.CociteConnections, flyConnection)
    else
        if flyConnection then
            flyConnection:Disconnect()
            flyConnection = nil
        end
        currentFlyCFrame = nil
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end
end

local isInternalRay = false

IsWallCheckVisible = function(part, targetChar)
    if not Settings.SilentAim.WallCheck then return true end
    local localChar = LocalPlayer.Character
    if not localChar then return false end

    local origin = Camera.CFrame.Position
    local dest = part.Position
    local dir = dest - origin

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {localChar, Camera}
    params.IgnoreWater = true

    isInternalRay = true
    local hit = workspace:Raycast(origin, dir, params)
    isInternalRay = false
    if not hit or hit.Instance:IsDescendantOf(targetChar) then
        return true
    end
    return false
end

            local args = {...}
            if checkcaller() then
                return oldRandom(...)
            end

            if Settings.NoSpread and Settings.NoSpread.Enabled then
                if (#args == 0) or
                   (args[1] == -0.05 and args[2] == 0.05) or
                   (args[1] == -0.1 and args[2] == 0.1) or
                   (args[1] == -0.1) or
                   (args[1] == -0.05) then

                end
            end

        end))

end)

local function CalcRayDirection(origin, targetPos, spread)
    local dir = (targetPos - origin).Unit
    if spread and spread > 0 then
        local spreadDist = spread * ((targetPos - origin).Magnitude / 100)
        dir = ((targetPos + Vector3.new(
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000
        )) - origin).Unit
    end
    return dir * 1000
end

local function GetGun()
    local char = LocalPlayer.Character
    if not char then return nil end
    return char:FindFirstChildOfClass("Tool")
end

local function RapidFireLoop()
    if rapidLoopRunning then return end
    rapidLoopRunning = true
    while rapidHolding and Settings.RapidFire.Enabled do
        local gun = GetGun()
        if gun then
            pcall(function() gun:Activate() end)
            -- спамим последний настоящий выстрел (те же аргументы, что послала игра)
            if lastShootSelf and lastShootArgs then
                rapidReplaying = true
                pcall(function()
                    lastShootSelf:FireServer(table.unpack(lastShootArgs))
                end)
                rapidReplaying = false
            end
        end
        task.wait(RAPID_DELAY)
    end
    rapidLoopRunning = false
end

local function StartRapid()
    if not Settings.RapidFire.Enabled or rapidHolding then return end
    rapidHolding = true
    FireHoldBtn.BackgroundColor3 = Color3.fromRGB(255, 70, 60)

end

    local bestTarget = nil
    local bestPart = nil
    local bestPos = nil

    if Settings.SilentAim.Enabled then
        if lockedSilentTarget and IsValidTarget(lockedSilentTarget) then
            local part = GetCharacterAimPart(lockedSilentTarget.Character, Settings.SilentAim.AimPart)
            if part then
                if not Settings.SilentAim.WallCheck or IsWallCheckVisible(part, lockedSilentTarget.Character) then
                    bestTarget = lockedSilentTarget
                    bestPart = part
                    bestPos = GetPredictedPosition(part, Settings.SilentAim.UsePrediction, Settings.SilentAim.Prediction)
                end
            end
        else
            if lockedSilentTarget then
                lockedSilentTarget = nil
                SilentFloatBtn.BackgroundColor3 = Color3.fromRGB(0, 140, 190)
                SilentFloatBtn.Text = "🔮 SILENT: ON"
            end
        end
    end

    currentSilentTarget = bestTarget
    currentSilentTargetPart = bestPart
    currentSilentAimPos = bestPos

    if AimbotFOVCircle then
        AimbotFOVCircle.Visible = Settings.Aimbot.Enabled and Settings.Aimbot.ShowFOV
        AimbotFOVCircle.Radius = Settings.Aimbot.FOV
        AimbotFOVCircle.Position = center
    end
    if SilentFOVCircle then
        SilentFOVCircle.Visible = Settings.SilentAim.Enabled and Settings.SilentAim.ShowFOV
        SilentFOVCircle.Radius = Settings.SilentAim.FOV
        SilentFOVCircle.Position = center
    end

    if SilentTracerLine then
        if Settings.SilentAim.Enabled and Settings.SilentAim.Tracers and currentSilentTargetPart then
            local screenPos, onScreen = Camera:WorldToViewportPoint(currentSilentTargetPart.Position)
            if onScreen and screenPos.Z > 0 then
                SilentTracerLine.From = center
                SilentTracerLine.To = Vector2.new(screenPos.X, screenPos.Y)
                SilentTracerLine.Visible = true
            else
                SilentTracerLine.Visible = false
            end
        else
            SilentTracerLine.Visible = false
        end
    end
end)

local Mouse = LocalPlayer:GetMouse()
pcall(function()
    if hookmetamethod and newcclosure and checkcaller then
        local oldIndex
        oldIndex = hookmetamethod(game, "__index", newcclosure(function(self, key)
            if not checkcaller() and (key == "Hit" or key == "hit" or key == "Target" or key == "target") and (self == Mouse) then
                if Settings.SilentAim.Enabled and currentSilentAimPos then
                    if key == "Hit" or key == "hit" then
                        return CFrame.new(currentSilentAimPos)
                    elseif (key == "Target" or key == "target") and currentSilentTargetPart then
                        return currentSilentTargetPart
                    end
                end
            end
            return oldIndex(self, key)
        end))
    end
end)

pcall(function()
    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            local self = args[1]

            if not checkcaller() and not isInternalRay then
                if method == "FireServer" and tostring(self) == "MainEvent" then
                    local eventArg = tostring(args[2] or args[1] or "")
                    local eventArgLower = string.lower(eventArg)

                    -- RAPID FIRE v2: ловим настоящий выстрел игрока
                    if eventArg == "ShootGun" or string.find(eventArgLower, "shoot", 1, true) then
                        if not rapidReplaying then
                            lastShootSelf = self
                            lastShootArgs = {}
                            for i = 2, #args do
                                lastShootArgs[i - 1] = args[i]
                            end
                        end
                        -- каждый реальный выстрел -> x3 при включённом Rapid
                        if Settings.RapidFire.Enabled then
                            for _ = 1, RAPID_MULT do
                                pcall(function()
                                    oldNamecall(self, table.unpack(args, 2))
                                end)
                            end
                        end
                    elseif Settings.SilentAim.Enabled and currentSilentAimPos then
                        if eventArg == "UpdateMousePos" or eventArg == "UpdateMousePosI" or string.find(eventArgLower, "mousepos") then
                            if args[2] == eventArg then
                                args[3] = currentSilentAimPos
                            else
                                args[2] = currentSilentAimPos
                            end
                            return oldNamecall(table.unpack(args))
                        end
                    end
                elseif Settings.SilentAim.Enabled and currentSilentAimPos
                    and self == workspace and method == "Raycast" then
                    if typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                        local origin = args[2]
                        local targetPos = currentSilentAimPos

                        local toolName = equippedToolName or ""
                        local isShotgun = string.find(string.lower(toolName), "double")
                                       or string.find(string.lower(toolName), "tactical")
                                       or string.find(string.lower(toolName), "shotgun")
                        local spreadVal = 0
                        if not (Settings.NoSpread and Settings.NoSpread.Enabled) then
                            spreadVal = isShotgun and 6 or 0
                        else
                            spreadVal = ((Settings.NoSpread.Amount or 0) / 100) * (isShotgun and 6 or 0)
                        end
                        args[3] = CalcRayDirection(origin, targetPos, spreadVal)
                        return oldNamecall(table.unpack(args))
                    end
                end
            end

            return oldNamecall(...)
        end))
    end
end)

RunService:BindToRenderStep("CociteCamLock", Enum.RenderPriority.Camera.Value + 1, function(dt)
    pcall(function()
        if Settings.Aimbot.Enabled and AimActive then
            local cam = workspace.CurrentCamera or Camera
            if not cam then return end

            local target = Settings.Aimbot.LockedTarget
            if target then
                if not IsValidTarget(target) then
                    target = nil
                    Settings.Aimbot.LockedTarget = nil
                else
                    local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
                    if part then
                        local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                        local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if not onScreen or screenPos.Z <= 0 or dist > (Settings.Aimbot.FOV * 1.5) then
                            target = nil
                            Settings.Aimbot.LockedTarget = nil
                        end
                    else
                        target = nil
                        Settings.Aimbot.LockedTarget = nil
                    end
                end
            end

            if not target then
                target = GetClosestPlayerForAim()
                Settings.Aimbot.LockedTarget = target
            end

            if target and target.Character then
                local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
                if part then
                    local aimPos = GetPredictedPosition(part, Settings.Aimbot.UsePrediction, Settings.Aimbot.Prediction)
                    if aimPos then
                        local camPos = cam.CFrame.Position
                        local targetCF = CFrame.lookAt(camPos, aimPos)
                        if Settings.Aimbot.Smooth then
                            local smoothness = math.clamp(Settings.Aimbot.Smoothness or 0.35, 0.01, 1)
                            local currentCF = cam.CFrame
                            cam.CFrame = currentCF:Lerp(targetCF, smoothness)
                        else
                            cam.CFrame = targetCF
                        end
                    end
                end
            end
        end
    end)
end)

-- ===================== ESP ENGINE =====================
local espObjects = {}

local function removeESPFor(player)
    local obj = espObjects[player]
    if obj then
        if obj.highlight then pcall(function() obj.highlight:Destroy() end) end
        if obj.box then pcall(function() obj.box:Remove() end) end
        if obj.name then pcall(function() obj.name:Remove() end) end
        espObjects[player] = nil
    end
end

local function clearAllESP()
    for player in pairs(espObjects) do
        removeESPFor(player)
    end
end

table.insert(_G.CociteConnections, Players.PlayerRemoving:Connect(removeESPFor))

table.insert(_G.CociteConnections, RunService.RenderStepped:Connect(function()
    if not ScreenGui.Parent then return end
    if not Settings.ESP.Enabled then
        clearAllESP()
        return
    end

    local cam = workspace.CurrentCamera or Camera
    if not cam then return end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local obj = espObjects[player]
            if not obj then
                espObjects[player] = {}
                if Drawing and Drawing.new then
                    local box = Drawing.new("Square")
                    box.Thickness = 1.5
                    box.Filled = false
                    box.Color = Color3.fromRGB(200, 90, 255)
                    box.Visible = false
                    local nameTxt = Drawing.new("Text")
                    nameTxt.Size = 13
                    nameTxt.Center = true
                    nameTxt.Outline = true
                    nameTxt.Color = Color3.fromRGB(235, 205, 255)
                    nameTxt.Visible = false
                    espObjects[player].box = box
                    espObjects[player].name = nameTxt
                end
                obj = espObjects[player]
            end

            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local alive = char and hum and hum.Health > 0 and hrp

            if alive and IsValidTarget(player) then
                if Settings.ESP.Chams then
                    local hl = obj.highlight
                    if not hl or hl.Parent ~= char then
                        if hl then pcall(function() hl:Destroy() end) end
                        local newHL = Instance.new("Highlight")
                        newHL.Name = "CociteESP_HL"
                        newHL.FillColor = Color3.fromRGB(120, 30, 210)
                        newHL.OutlineColor = Color3.fromRGB(235, 160, 255)
                        newHL.FillTransparency = 0.6
                        newHL.OutlineTransparency = 0
                        newHL.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        newHL.Parent = char
                        obj.highlight = newHL
                    end
                elseif obj.highlight then
                    pcall(function() obj.highlight:Destroy() end)
                    obj.highlight = nil
                end

                local rootPos, onScreen = cam:WorldToViewportPoint(hrp.Position)
                if onScreen and rootPos.Z > 0 then
                    local cf, sz = char:GetBoundingBox()
                    local minX, minY = math.huge, math.huge
                    local maxX, maxY = -math.huge, -math.huge
                    local ok = true
                    for x = -1, 1, 2 do
                        for y = -1, 1, 2 do
                            for z = -1, 1, 2 do
                                local world = (cf * CFrame.new(sz.X / 2 * x, sz.Y / 2 * y, sz.Z / 2 * z)).Position
                                local sp, vis = cam:WorldToViewportPoint(world)
                                if vis and sp.Z > 0 then
                                    if sp.X < minX then minX = sp.X end
                                    if sp.X > maxX then maxX = sp.X end
                                    if sp.Y < minY then minY = sp.Y end
                                    if sp.Y > maxY then maxY = sp.Y end
                                else
                                    ok = false
                                end
                            end
                        end
                    end

                    if ok then
                        if obj.box then
                            if Settings.ESP.Boxes then
                                obj.box.Size = Vector2.new(maxX - minX, maxY - minY)
                                obj.box.Position = Vector2.new(minX, minY)
                                obj.box.Visible = true
                            else
                                obj.box.Visible = false
                            end
                        end
                        if obj.name then
                            if Settings.ESP.Names then
                                obj.name.Text = player.Name .. " [" .. math.floor(hum.Health) .. "]"
                                obj.name.Position = Vector2.new((minX + maxX) / 2, minY - 16)
                                obj.name.Visible = true
                            else
                                obj.name.Visible = false
                            end
                        end
                    else
                        if obj.box then obj.box.Visible = false end
                        if obj.name then obj.name.Visible = false end
                    end
                else
                    if obj.box then obj.box.Visible = false end
                    if obj.name then obj.name.Visible = false end
                end
            else
                if obj.highlight then
                    pcall(function() obj.highlight:Destroy() end)
                    obj.highlight = nil
                end
                if obj.box then obj.box.Visible = false end
                if obj.name then obj.name.Visible = false end
            end
        end
    end
end))
