local function GetClosestToCenter()
    local closestP = nil
    local shortestDist = Settings.Aimbot.FOV
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            local part = player.Character:FindFirstChild("UpperTorso") or player.Character:FindFirstChild("HumanoidRootPart")
            if part then
                local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local magnitude = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if magnitude < shortestDist then
                        shortestDist = magnitude
                        closestP = player
                    end
                end
            end
        end
    end
    return closestP
end

local AimActive = false
AimFloatBtn.MouseButton1Click:Connect(function()
    AimActive = not AimActive
    if AimActive then
        AimFloatBtn.BackgroundColor3 = Color3.fromRGB(110, 25, 200)
        AimFloatBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        AimFloatBtn.Text = "🎯 AIM: ON"
        Settings.Aimbot.LockedTarget = GetClosestToCenter()
    else
        AimFloatBtn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
        AimFloatBtn.TextColor3 = Color3.fromRGB(200, 150, 255)
        AimFloatBtn.Text = "🎯 AIM: OFF"
        Settings.Aimbot.LockedTarget = nil
    end
end)

RunService.RenderStepped:Connect(function()
    if Settings.Aimbot.Enabled and AimActive then
        local target = Settings.Aimbot.LockedTarget
        if target and target.Character and target.Character:FindFirstChild("Humanoid") and target.Character.Humanoid.Health > 0 then
            local targetPart = target.Character:FindFirstChild("UpperTorso") or target.Character:FindFirstChild("HumanoidRootPart")
            if targetPart then
                -- Жесткий моментальный лок (без плавности)
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPart.Position)
            end
        else
            Settings.Aimbot.LockedTarget = GetClosestToCenter()
        end
    end
end)

local function CreateButton(name, posY, callback)
    local btn = Instance.new("TextButton", MainFrame)
    btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
    btn.Position = UDim2.new(0, 20, 0, posY)
    btn.Size = UDim2.new(0, 320, 0, 45)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = name .. ": OFF"
    btn.TextColor3 = Color3.fromRGB(190, 180, 210)
    btn.TextSize = 13
    
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(90, 40, 160)
    stroke.Thickness = 1.2
    
    local active = false
    btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(110, 25, 200)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = name .. ": ON"
        else
            btn.BackgroundColor3 = Color3.fromRGB(20, 15, 32)
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = name .. ": OFF"
        end
        callback(active)
    end)
    return btn
end

-- 5. ESP
-------------------------------------------------------------------
local function ClearESP()
    for _, objList in pairs(Settings.ESP.ActiveDrawings) do
        for _, obj in pairs(objList) do
            if obj then obj:Remove() end
        end
    end
    Settings.ESP.ActiveDrawings = {}
end

local function ToggleESP(state)
    Settings.ESP.Enabled = state
    if not state then
        ClearESP()
    end
end

Players.PlayerRemoving:Connect(function(player)
    if Settings.ESP.ActiveDrawings[player] then
        for _, obj in pairs(Settings.ESP.ActiveDrawings[player]) do
            if obj then obj:Remove() end
        end
        Settings.ESP.ActiveDrawings[player] = nil
    end
end)

RunService.RenderStepped:Connect(function()
    if not Settings.ESP.Enabled then return end
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local humanoid = char and char:FindFirstChild("Humanoid")
            
            if hrp and humanoid and humanoid.Health > 0 then
                local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                
                if not Settings.ESP.ActiveDrawings[player] then
                    local box = Drawing.new("Square")
                    box.Visible = false
                    box.Color = Color3.fromRGB(180, 50, 255)
                    box.Thickness = 1.5
                    box.Filled = false
                    
                    local nameText = Drawing.new("Text")
                    nameText.Visible = false
                    nameText.Color = Color3.fromRGB(255, 255, 255)
                    nameText.Size = 13
                    nameText.Center = true
                    nameText.Outline = true
                    
                    Settings.ESP.ActiveDrawings[player] = {Box = box, Name = nameText}
                end
                
                local visuals = Settings.ESP.ActiveDrawings[player]
                local box, nameText = visuals.Box, visuals.Name
                
                if onScreen then
                    local scaleFactor = 300 / pos.Z
                    local size = Vector2.new(2, 3) * scaleFactor
                    box.Size = size
                    box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                    box.Visible = true
                    
                    nameText.Text = player.Name
                    nameText.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 18)
                    nameText.Visible = true
                else
                    box.Visible = false
                    nameText.Visible = false
                end
            else
                if Settings.ESP.ActiveDrawings[player] then
                    Settings.ESP.ActiveDrawings[player].Box.Visible = false
                    Settings.ESP.ActiveDrawings[player].Name.Visible = false
                end
            end
        end
    end
end)
