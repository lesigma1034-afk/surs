local friendCache = {}
local function IsFriendCached(userId)
    if not userId then return false end
    if friendCache[userId] ~= nil then
        return friendCache[userId]
    end
    task.spawn(function()
        local f = false
        pcall(function()
            f = LocalPlayer:IsFriendsWith(userId)
        end)
        friendCache[userId] = f
    end)
    return false
end

local function IsWhitelisted(player)
    if not player then return false end
    if Settings.Whitelist.Players[player.Name]
       or Settings.Whitelist.Players[player.UserId]
       or Settings.Whitelist.Players[tostring(player.UserId)]
       or Settings.Whitelist.Players[string.lower(player.Name)] then
        return true
    end
    if Settings.Whitelist.Team and LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
        return true
    end
    if Settings.Whitelist.Friends then
        if IsFriendCached(player.UserId) then return true end
    end
    return false
end

local function IsValidTarget(player)
    if player == LocalPlayer or not player.Character then return false end
    if IsWhitelisted(player) then return false end
    local char = player.Character
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not hrp then return false end

    local bodyEffects = char:FindFirstChild("BodyEffects")
    if bodyEffects then
        local ko = bodyEffects:FindFirstChild("K.O") or bodyEffects:FindFirstChild("KO")
        if ko and ko.Value == true then
            return false
        end
    end

    return true
end

local function GetCharacterAimPart(character, chosenPart)
    if not character then return nil end
    local partName = chosenPart or "HumanoidRootPart"
    if partName == "Random" then
        local parts = {"Head", "UpperTorso", "HumanoidRootPart"}
        partName = parts[math.random(1, #parts)]
    end
    local part = character:FindFirstChild(partName)
    if not part and (partName == "Head" or partName == "UpperTorso") then
        part = character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
    return part or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
end

local function GetPredictedPosition(part, usePred, predAmount)
    if not part then return nil end
    local pos = part.Position
    if usePred then
        local char = part.Parent
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local vel = (hrp and (hrp.AssemblyLinearVelocity or hrp.Velocity)) or part.AssemblyLinearVelocity or part.Velocity
        if vel then
            pos = pos + (vel * (predAmount or 0.135))
        end
    end
    return pos
end

local function GetClosestPlayerForAim()
    local cam = workspace.CurrentCamera or Camera
    if not cam then return nil end
    local closest = nil
    local shortestDist = Settings.Aimbot.FOV or 220
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) then
            local part = GetCharacterAimPart(player.Character, Settings.Aimbot.AimPart)
            if part then
                local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        closest = player
                    end
                end
            end
        end
    end
    return closest
end

local function CreateToggle(parent, text, defaultState, callback)
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = defaultState and Color3.fromRGB(120, 30, 210) or Color3.fromRGB(18, 14, 30)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = text .. (defaultState and ": ON" or ": OFF")
    btn.TextColor3 = defaultState and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 170)
    stroke.Thickness = 1.3

    local active = defaultState
    btn.MouseButton1Click:Connect(function()
        active = not active
        if active then
            btn.BackgroundColor3 = Color3.fromRGB(120, 30, 210)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = text .. ": ON"
        else
            btn.BackgroundColor3 = Color3.fromRGB(18, 14, 30)
            btn.TextColor3 = Color3.fromRGB(190, 180, 210)
            btn.Text = text .. ": OFF"
        end
        callback(active)
    end)

    return {
        Button = btn,
        Set = function(self, val)
            active = val
            btn.BackgroundColor3 = active and Color3.fromRGB(120, 30, 210) or Color3.fromRGB(18, 14, 30)
            btn.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(190, 180, 210)
            btn.Text = text .. (active and ": ON" or ": OFF")
            callback(active)
        end
    }
end

local function CreateCycle(parent, label, options, defaultIndex, callback)
    local idx = defaultIndex or 1
    local btn = Instance.new("TextButton", parent)
    btn.BackgroundColor3 = Color3.fromRGB(18, 14, 30)
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.Font = Enum.Font.GothamSemibold
    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
    btn.TextColor3 = Color3.fromRGB(210, 190, 255)
    btn.TextSize = 11

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(100, 45, 170)
    stroke.Thickness = 1.3

    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
        callback(options[idx])
    end)

    return {
        Button = btn,
        Get = function() return options[idx] end,
        Set = function(self, val)
            for i, opt in ipairs(options) do
                if opt == val then
                    idx = i
                    btn.Text = label .. ": [" .. tostring(options[idx]) .. "]"
                    callback(options[idx])
                    break
                end
            end
        end
    }
end

        end
    end

local isInternalRay = false

IsWallCheckVisible = function(part, targetChar)
    if not Settings.SilentAim.WallCheck then return true end
    local localChar = LocalPlayer.Character
    if not localChar then return false end

    local origin = Camera.CFrame.Position
    local dest = part.Position
    local dir = dest - origin

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {localChar, Camera}
    params.IgnoreWater = true

    isInternalRay = true
    local hit = workspace:Raycast(origin, dir, params)
    isInternalRay = false
    if not hit or hit.Instance:IsDescendantOf(targetChar) then
        return true
    end
    return false
end

pcall(function()
    if hookfunction and newcclosure and checkcaller then
        local oldRandom
        oldRandom = hookfunction(math.random, newcclosure(function(...)
            local args = {...}
            if checkcaller() then
                return oldRandom(...)
            end

            if Settings.NoSpread and Settings.NoSpread.Enabled then
                if (#args == 0) or
                   (args[1] == -0.05 and args[2] == 0.05) or
                   (args[1] == -0.1 and args[2] == 0.1) or
                   (args[1] == -0.1) or
                   (args[1] == -0.05) then
                    local spread = Settings.NoSpread.Amount or 0
                    return oldRandom(...) * (spread / 100)
                end
            end

            return oldRandom(...)
        end))
    end
end)

local function CalcRayDirection(origin, targetPos, spread)
    local dir = (targetPos - origin).Unit
    if spread and spread > 0 then
        local spreadDist = spread * ((targetPos - origin).Magnitude / 100)
        dir = ((targetPos + Vector3.new(
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000,
            math.random(-spreadDist * 1000, spreadDist * 1000) / 1000
        )) - origin).Unit
    end
    return dir * 1000
end

-- ===================== RAPID FIRE v2 (hold to shoot fast) =====================
-- Логика:
--   1) держим ЛКМ / палец на экране -> каждый Delay вызываем Tool:Activate()
--      (Activate запускает штатный Tool.Activated, поэтому сервер видит нормальный выстрел)
--   2) если у gun есть ShootingCooldown / FireRate / Cooldown / FireDelay (числовые Value),
--      обнуляем их, из-за этого пушка реально стреляет без пауз
--   3) опционально дублируем выстрел через remote (Shoot / ShootGun), если он есть в тулзе
--   4) не стреляем из gun, который просто лежит в Backpack (иначе Activate уходит в никуда)

local rapidAccum = 0          -- накопитель времени (не зависит от fps и от task.wait-дрейфа)
local is_firing = false       -- текущее "зажато" состояние
local fireHeld = false        -- виртуальный флаг зажатия (мышь + тач)
local mouseHeld = false
local touchHeld = false
local rapidToggleObj = nil

--[[ Проверяем, похож ли Tool на пушку.
     Да Худ / Флейм Худ: у gun обычно есть Ammo + Handle (для стволов), либо ShootingCooldown.
     Но встречаются и кастомные стволы без Ammo, так что падаем на Handle + RemoteEvent. ]]
local function IsGunTool(tool)
    if not tool or not tool:IsA("Tool") then return false end
    if tool:FindFirstChild("Ammo") then return true end
    if tool:FindFirstChild("ShootingCooldown") then return true end
    if tool:FindFirstChild("FireRate") then return true end
    if tool:FindFirstChild("Handle") and tool:FindFirstChildWhichIsA("RemoteEvent") then return true end
    return false
end

--[[ Берём именно экипированный gun (в Character), а не из Backpack.
     Иначе Activate() уходит в пустоту: Activate работает только на заекипированной тулзе. ]]
local function GetGun()
    local char = LocalPlayer.Character
    if not char then return nil end
    for _, tool in ipairs(char:GetChildren()) do
        if IsGunTool(tool) then
            return tool
        end
    end
    return nil
end

--[[ Достаём патроны: Ammo может быть NumberValue, IntValue или Value, где угодно в тулзе. ]]
local function GetAmmoValue(tool)
    if not tool then return nil end
    local ammo = tool:FindFirstChild("Ammo")
    if not ammo then
        ammo = tool:FindFirstChild("Ammo", true)
    end
    if ammo and (ammo:IsA("NumberValue") or ammo:IsA("IntValue") or ammo:IsA("ValueBase")) then
        return ammo
    end
    return nil
end

--[[ Обнуление кулдаунов. Именно это делает rapid fire настоящим, а не косметическим.
     Много где cooldown хранится как Value-объект внутри тулзы или в внутренней папке.
     FindFirstChild(name, true) — рекурсивно, поэтому достаёт и вложенные значения. ]]
local COOLDOWN_NAMES = {
    "ShootingCooldown", "FireRate", "Cooldown", "FireDelay",
    "ShootCooldown", "Rate", "ReloadCooldown"
}

local function ZeroGunCooldowns(tool)
    if not tool or not Settings.RapidFire.ZeroCooldown then return end
    for _, name in ipairs(COOLDOWN_NAMES) do
        local v = tool:FindFirstChild(name, true)
        if v and v:IsA("ValueBase") and typeof(v.Value) == "number" and v.Value ~= 0 then
            pcall(function() v.Value = 0 end)
        end
    end
end

--[[ Пройтись по ВСЕМ пушкам: и в руках, и в рюкзаке.
     Смысл: gun в Backpack тоже держит свой кулдаун. Если обнулить заранее,
     то после переключения ствола он уже стреляет быстро, а не ждёт тик. ]]
local function ZeroAllGunCooldowns()
    if not Settings.RapidFire.ZeroCooldown then return end
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, tool in ipairs(backpack:GetChildren()) do
            if IsGunTool(tool) then
                ZeroGunCooldowns(tool)
            end
        end
    end
    local char = LocalPlayer.Character
    if char then
        for _, tool in ipairs(char:GetChildren()) do
            if IsGunTool(tool) then
                ZeroGunCooldowns(tool)
            end
        end
    end
end

--[[ Опциональный дубль выстрела через remote. По умолчанию выключено, чтобы не спалить акк
     и не задвоить урон. Включается тумблером UseRemote, если сервер разрешает.  ]]
local function FireRemoteShot(tool)
    if not tool or not Settings.RapidFire.UseRemote then return end
    local remotes = { "Shoot", "ShootGun", "Fire", "FireGun", "MainEvent" }
    for _, name in ipairs(remotes) do
        local remote = tool:FindFirstChild(name)
        if remote and remote:IsA("RemoteEvent") then
            pcall(function()
                if name == "MainEvent" then
                    remote:FireServer("Shoot")
                else
                    remote:FireServer()
                end
            end)
            return
        end
    end
end

--[[ Одна итерация выстрела. Возвращает true/false — стрельнули ли. ]]
local function RapidFireStep()
    local gun = GetGun()
    if not gun then return false end

    -- Игры часто гасят Tool.Enabled на время кулдауна. Если включён ZeroCooldown,
    -- возвращаем тулзу в рабочее состояние, иначе просто ждём.
    if not gun.Enabled then
        if Settings.RapidFire.ZeroCooldown then
            pcall(function() gun.Enabled = true end)
        else
            return false
        end
    end

    local ammo = GetAmmoValue(gun)
    if ammo then
        if Settings.RapidFire.RequireAmmo and ammo.Value <= 0 then
            -- патронов нет: пробуем перезарядить штатным способом, если есть
            local reload = gun:FindFirstChild("Reload") or gun:FindFirstChild("ReloadEvent")
            if reload and reload:IsA("RemoteEvent") then
                pcall(function() reload:FireServer() end)
            end
            return false
        end
        if Settings.RapidFire.InfAmmo then
            pcall(function() ammo.Value = math.max(ammo.Value, 1) end)
        end
    end

    ZeroGunCooldowns(gun)

    -- основной путь: штатная активация (Tool.Activated -> серверный выстрел)
    pcall(function() gun:Activate() end)
    FireRemoteShot(gun)
    return true
end

--[[ Главный цикл. Обновляется из Heartbeat, из-за этого не зависит от fps и не копит дрейф.
     delay = 0 -> стреляем каждый кадр, delay > 0 -> держим ровный интервал. ]]
local function RapidFireTick(dt)
    if not Settings.RapidFire.Enabled or not fireHeld then
        if is_firing then
            is_firing = false
        end
        rapidAccum = 0
        return
    end

    local delay = Settings.RapidFire.Delay or 0.01
    if delay < 0 then delay = 0 end

    rapidAccum = rapidAccum + dt
    if rapidAccum < delay then return end

    -- вычитаем, а не сбрасываем в 0: так интервал не "квантуется" по кадрам
    -- (при 60 fps и delay=0.02 это разница между 30 и 50 выстрелами в секунду)
    rapidAccum = rapidAccum - delay
    if rapidAccum > delay then rapidAccum = 0 end -- защита от "очереди" после лага

    is_firing = RapidFireStep()
end

--[[ Не начинаем стрелять, если клик попал по ИНТЕРАКТИВНОМУ элементу интерфейса
     (кнопка нашего хаба, топбар, чат). Обычные декоративные фреймы/лейблы не считаем,
     иначе невидимая полноэкранная панель могла бы заблокировать стрельбу целиком.
     Основной фильтр всё равно gameProcessedEvent — это лишь перестраховка. ]]
local INTERACTIVE_CLASSES = {
    TextButton = true,
    ImageButton = true,
    TextBox = true,
    ScrollingFrame = true,
}

local function IsOverGui(x, y)
    local hx, hy = math.floor(x), math.floor(y)
    local blocked = false

    local function scan(container)
        if blocked or not container then return end
        local ok, list = pcall(function()
            return container:GetGuiObjectsAtPosition(hx, hy)
        end)
        if not ok or not list then return end
        for _, obj in ipairs(list) do
            -- кнопки хаба — это Frame+TextButton; ловим и сам объект, и его кнопку-ребёнка
            if INTERACTIVE_CLASSES[obj.ClassName] then
                blocked = true
                return
            end
            for _, child in ipairs(obj:GetChildren()) do
                if INTERACTIVE_CLASSES[child.ClassName] then
                    blocked = true
                    return
                end
            end
        end
    end

    pcall(function() scan(game:GetService("CoreGui")) end)
    pcall(function() scan(LocalPlayer:FindFirstChildOfClass("PlayerGui")) end)
    pcall(function() scan(ScreenGui) end)

    return blocked
end

--[[ Тач-стрельба: считаем "зажатием" удержание пальца на экране.
     Тут только трекинг самих тачей — сам выстрел живёт в RapidFireTick. ]]
local TouchHeldCount = 0

table.insert(_G.CociteConnections, UserInputService.InputBegan:Connect(function(input, gpe)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        if gpe then return end
        local pos = UserInputService:GetMouseLocation()
        if IsOverGui(pos.X, pos.Y) then return end
        mouseHeld = true
        fireHeld = true
    elseif input.UserInputType == Enum.UserInputType.Touch then
        if gpe then return end
        local pos = input.Position
        if IsOverGui(pos.X, pos.Y) then return end
        TouchHeldCount = TouchHeldCount + 1
        touchHeld = true
        fireHeld = true
    end
end))

table.insert(_G.CociteConnections, UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        mouseHeld = false
        if TouchHeldCount <= 0 then
            fireHeld = false
        end
    elseif input.UserInputType == Enum.UserInputType.Touch then
        TouchHeldCount = math.max(0, TouchHeldCount - 1)
        if TouchHeldCount <= 0 then
            touchHeld = false
            if not mouseHeld then
                fireHeld = false
            end
        end
    end
end))

-- страховка: если по какой-то причине InputEnded потерялся (свёрнутое окно, дисконнект),
-- проверяем реальное состояние кнопки/пальцев и сбрасываем флаг
table.insert(_G.CociteConnections, UserInputService.WindowFocusReleased:Connect(function()
    mouseHeld = false
    if TouchHeldCount <= 0 then
        fireHeld = false
    end
end))

-- стабильный тик: Heartbeat не привязан к fps камеры и не копит дрейф
local cooldownSweepAccum = 0
local rapidTickConn = RunService.Heartbeat:Connect(function(dt)
    RapidFireTick(dt)

    -- раз в 0.25 сек проходим по всем пушкам (включая рюкзак) и держим кулдаун на нуле.
    -- Так работает даже без зажатия ЛКМ: ствол уже "горячий" к первому выстрелу.
    if Settings.RapidFire.Enabled and Settings.RapidFire.ZeroCooldown then
        cooldownSweepAccum = cooldownSweepAccum + dt
        if cooldownSweepAccum >= 0.25 then
            cooldownSweepAccum = 0
            ZeroAllGunCooldowns()
        end
    end
end)
table.insert(_G.CociteConnections, rapidTickConn)

    if hookmetamethod and getnamecallmethod and newcclosure and checkcaller then
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local method = getnamecallmethod()
            local args = {...}
            local self = args[1]

            if not checkcaller() and not isInternalRay and Settings.SilentAim.Enabled and currentSilentAimPos then
                if self == workspace and method == "Raycast" then
                    if typeof(args[2]) == "Vector3" and typeof(args[3]) == "Vector3" then
                        local origin = args[2]
                        local targetPos = currentSilentAimPos

                        local toolName = equippedToolName or ""
                        local isShotgun = string.find(string.lower(toolName), "double")
                                       or string.find(string.lower(toolName), "tactical")
                                       or string.find(string.lower(toolName), "shotgun")
                        local spreadVal = 0
                        if not (Settings.NoSpread and Settings.NoSpread.Enabled) then
                            spreadVal = isShotgun and 6 or 0
                        else
                            spreadVal = ((Settings.NoSpread.Amount or 0) / 100) * (isShotgun and 6 or 0)
                        end
                        args[3] = CalcRayDirection(origin, targetPos, spreadVal)
                        return oldNamecall(table.unpack(args))
                    end
                elseif method == "FireServer" and tostring(self) == "MainEvent" then
                    local eventArg = tostring(args[2] or args[1] or "")
                    if eventArg == "UpdateMousePos" or eventArg == "UpdateMousePosI" or string.find(string.lower(eventArg), "mousepos") then
                        if args[2] == eventArg then
                            args[3] = currentSilentAimPos
                        else
                            args[2] = currentSilentAimPos
                        end
                        return oldNamecall(table.unpack(args))
                    end
                end
            end

            return oldNamecall(...)
        end))
    end
end)

RunService:BindToRenderStep("CociteCamLock", Enum.RenderPriority.Camera.Value + 1, function(dt)
    pcall(function()
        if Settings.Aimbot.Enabled and AimActive then
            local cam = workspace.CurrentCamera or Camera
            if not cam then return end

            local target = Settings.Aimbot.LockedTarget
            if target then
                if not IsValidTarget(target) then
                    target = nil
                    Settings.Aimbot.LockedTarget = nil
                else
                    local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
                    if part then
                        local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                        local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if not onScreen or screenPos.Z <= 0 or dist > (Settings.Aimbot.FOV * 1.5) then
                            target = nil
                            Settings.Aimbot.LockedTarget = nil
                        end
                    else
                        target = nil
                        Settings.Aimbot.LockedTarget = nil
                    end
                end
            end

            if not target then
                target = GetClosestPlayerForAim()
                Settings.Aimbot.LockedTarget = target
            end

            if target and target.Character then
                local part = GetCharacterAimPart(target.Character, Settings.Aimbot.AimPart)
                if part then
                    local aimPos = GetPredictedPosition(part, Settings.Aimbot.UsePrediction, Settings.Aimbot.Prediction)
                    if aimPos then
                        local camPos = cam.CFrame.Position
                        local targetCF = CFrame.lookAt(camPos, aimPos)
                        if Settings.Aimbot.Smooth then
                            local smoothness = math.clamp(Settings.Aimbot.Smoothness or 0.35, 0.01, 1)
                            local smoothFactor = math.clamp(1 - math.pow(1 - math.clamp(smoothness, 0.01, 0.99), (dt or 0.016) * 60), 0.01, 1)
                            cam.CFrame = cam.CFrame:Lerp(targetCF, smoothFactor)
                        else
                            cam.CFrame = targetCF
                        end
                    end
                end
            end
        end
    end)
end)

listContainer.Size = UDim2.new(1, 0, 0, 0)
listContainer.AutomaticSize = Enum.AutomaticSize.Y

listLayout.SortOrder = Enum.SortOrder.LayoutOrder
listLayout.Padding = UDim.new(0, 4)

local function UpdateWhitelistUI()
    for _, child in ipairs(listContainer:GetChildren()) do
        if child:IsA("Frame") or child:IsA("TextLabel") or child:IsA("TextButton") then
            child:Destroy()
        end
    end

    local count = 0
    for name, _ in pairs(Settings.Whitelist.Players) do
        count = count + 1
        local itemFrame = Instance.new("Frame", listContainer)
        itemFrame.BackgroundColor3 = Color3.fromRGB(22, 16, 34)
        itemFrame.Size = UDim2.new(1, 0, 0, 28)
        Instance.new("UICorner", itemFrame).CornerRadius = UDim.new(0, 8)
        local itemStroke = Instance.new("UIStroke", itemFrame)
        itemStroke.Color = Color3.fromRGB(80, 40, 140)
        itemStroke.Thickness = 1

        local nameLabel = Instance.new("TextLabel", itemFrame)
        nameLabel.BackgroundTransparency = 1
        nameLabel.Position = UDim2.new(0, 10, 0, 0)
        nameLabel.Size = UDim2.new(1, -40, 1, 0)
        nameLabel.Font = Enum.Font.GothamSemibold
        nameLabel.Text = "🛡️ " .. tostring(name)
        nameLabel.TextColor3 = Color3.fromRGB(220, 200, 255)
        nameLabel.TextSize = 11
        nameLabel.TextXAlignment = Enum.TextXAlignment.Left

        local delBtn = Instance.new("TextButton", itemFrame)
        delBtn.BackgroundColor3 = Color3.fromRGB(200, 40, 60)
        delBtn.Position = UDim2.new(1, -26, 0.5, -9)
        delBtn.Size = UDim2.new(0, 18, 0, 18)
        delBtn.Font = Enum.Font.GothamBold
        delBtn.Text = "✕"
        delBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        delBtn.TextSize = 10
        Instance.new("UICorner", delBtn).CornerRadius = UDim.new(0, 4)

        delBtn.MouseButton1Click:Connect(function()
            Settings.Whitelist.Players[name] = nil
            Settings.Whitelist.Players[string.lower(name)] = nil
            UpdateWhitelistUI()
        end)
    end

    if count == 0 then
        local emptyLbl = Instance.new("TextLabel", listContainer)
        emptyLbl.BackgroundTransparency = 1
        emptyLbl.Size = UDim2.new(1, 0, 0, 24)
        emptyLbl.Font = Enum.Font.Gotham
        emptyLbl.Text = "No custom players whitelisted"
        emptyLbl.TextColor3 = Color3.fromRGB(120, 110, 140)
        emptyLbl.TextSize = 11
    end
end

addPickerBtn.MouseButton1Click:Connect(function()
    if selectedPlayerName and selectedPlayerName ~= "" and selectedPlayerName ~= "None in server" then
        Settings.Whitelist.Players[selectedPlayerName] = true
        Settings.Whitelist.Players[string.lower(selectedPlayerName)] = true
        UpdateWhitelistUI()
    end
end)

addManualBtn.MouseButton1Click:Connect(function()
    local text = manualBox.Text:gsub("%s+", "")
    if text ~= "" then
        Settings.Whitelist.Players[text] = true
        Settings.Whitelist.Players[string.lower(text)] = true
        manualBox.Text = ""
        UpdateWhitelistUI()
    end
end)
